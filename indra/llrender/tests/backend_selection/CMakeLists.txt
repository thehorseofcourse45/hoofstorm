cmake_minimum_required(VERSION 3.16)
project(backend_selection_tests LANGUAGES CXX)
enable_testing()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/../../llrenderbackend.cpp")
# Compile the production factory bodies against inert API objects, without needing a GL context.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../llrenderbackend.cpp" factories)
string(REGEX REPLACE "#include[^\n]*" "" factories "${factories}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/factory_bodies.cpp"
    "#include \"factory_test_stubs.h\"\n#include \"llrenderbackendselection.h\"\n${factories}")

foreach(variant runtime_on runtime_off)
    add_executable(${variant} selection_test.cpp ../../llrenderbackendselection.cpp)
    target_include_directories(${variant} PRIVATE ../..)
    target_compile_features(${variant} PRIVATE cxx_std_17)
    if(variant STREQUAL "runtime_on")
        target_compile_definitions(${variant} PRIVATE LL_VULKAN_RUNTIME_AVAILABLE=1)
    endif()
    if(MSVC)
        target_compile_options(${variant} PRIVATE /W4 /WX)
    endif()
    add_test(NAME ${variant} COMMAND ${variant})
    add_test(NAME ${variant}_marker COMMAND ${variant} marker)
    add_test(NAME ${variant}_marker_retry COMMAND ${variant} retry)
endforeach()
foreach(variant runtime_on runtime_off)
    add_executable(factory_${variant} factory_test.cpp
        "${CMAKE_CURRENT_BINARY_DIR}/factory_bodies.cpp" ../../llrenderbackendselection.cpp)
    target_include_directories(factory_${variant} PRIVATE ../.. "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_features(factory_${variant} PRIVATE cxx_std_17)
    if(variant STREQUAL "runtime_on")
        target_compile_definitions(factory_${variant} PRIVATE LL_VULKAN_RUNTIME_AVAILABLE=1)
    endif()
    if(MSVC)
        target_compile_options(factory_${variant} PRIVATE /W4 /WX)
    endif()
    foreach(first RANGE 0 4)
        add_test(NAME factory_${variant}_${first} COMMAND factory_${variant} ${first})
    endforeach()
endforeach()
